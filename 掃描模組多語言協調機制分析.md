## 📊 核心發現

### ✅ 掃描模組的成功案例

**掃描模組**使用**適配器模式**成功協調四語言引擎：
- **Python 引擎**: 原生 `import` 調用
- **TypeScript 引擎**: `subprocess` 調用 Node.js
- **Rust 引擎**: FFI Bridge + 線程池包裝
- **Go 引擎**: `subprocess` 調用編譯後的二進制

**關鍵成功因素**:
1. ✅ **統一接口**: `BaseScannerAdapter` 抽象基類
2. ✅ **依賴注入**: Coordinator 不依賴具體實現
3. ✅ **標準化輸出**: 所有引擎返回統一的 `Asset` 對象
4. ✅ **異步優先**: 所有方法都是 `async/await`
5. ✅ **優雅降級**: 引擎不可用時返回空結果而非崩潰

## 🏗️ 架構設計：適配器模式

### 1. 統一接口定義

**位置**: `services/scan/coordinators/engines/base.py`

```python
class BaseScannerAdapter(ABC):
    """掃描引擎適配器基類
    
    所有引擎適配器必須實現此接口，確保 Coordinator 可以統一調用。
    """
    
    @abstractmethod
    async def is_available(self) -> bool:
        """檢查引擎是否可用"""
        pass
    
    @abstractmethod
    async def scan(
        self,
        targets: List[str],
        options: Dict[str, Any]
    ) -> Dict[str, Any]:
        """執行掃描任務的統一入口
        
        Returns:
            Dict 包含：
            - assets: List[Asset] (標準化資產列表)
            - metadata: Dict (引擎元數據)
            - error: Optional[str] (錯誤訊息)
        """
        pass
    
    def _standardize_assets(
        self,
        raw_assets: List[Any],
        engine_name: str
    ) -> List[Asset]:
        """將引擎特定格式轉換為標準 Asset"""
        pass
```

**核心設計**:
- ✅ **異步優先**: 所有方法都是 `async`，避免阻塞
- ✅ **統一返回格式**: `{"assets": [...], "metadata": {...}, "error": None}`
- ✅ **標準化工具**: `_standardize_assets()` 輔助方法

```python
class PythonAdapter(BaseScannerAdapter):
    """Python 引擎適配器 - 原生調用"""
    
    async def is_available(self) -> bool:
        """檢查 Python 引擎是否可用"""
        try:
            # 嘗試導入 ScanOrchestrator
            from services.scan.engines.python_engine.scan_orchestrator import ScanOrchestrator
            return True
        except ImportError as e:
            self.logger.warning(f"Python 引擎不可用: {e}")
            return False
    
    async def scan(
        self,
        targets: List[str],
        options: Dict[str, Any]
    ) -> Dict[str, Any]:
        """執行 Python 引擎掃描"""
        # 1. 導入 ScanOrchestrator
        from services.scan.engines.python_engine.scan_orchestrator import ScanOrchestrator
        
        if self._orchestrator is None:
            self._orchestrator = ScanOrchestrator()
        
        # 2. 構建標準 Payload
        request = ScanStartPayload(
            scan_id=options.get("scan_id", "unknown"),
            targets=validated_urls,
            strategy=options.get("strategy", "deep"),
            custom_headers=options.get("headers", {})
        )
        
        # 3. 執行掃描
        result = await self._orchestrator.execute_scan(request)
        
        # 4. 標準化輸出
        return {
            "assets": result.assets,
            "metadata": {
                "engine": "python",
                "scan_duration": result.summary.scan_duration_seconds,
                "pages_crawled": result.summary.pages_crawled
            },
            "error": None
        }
```

**通信方式**: 
- ✅ **原生 Python 調用**（最快，零開銷）
- ✅ **直接訪問 Python 生態系統**（BeautifulSoup, Selenium）

#### 2.2 TypeScript Adapter - 子進程調用

**位置**: `services/scan/coordinators/engines/typescript_adapter.py`

```python
class TypeScriptAdapter(BaseScannerAdapter):
    """TypeScript 引擎適配器 - 子進程調用"""
    
    async def is_available(self) -> bool:
        """檢查 TypeScript 引擎是否可用"""
        # 1. 檢查編譯後的 JS 文件是否存在
        scanner_path = Path(__file__).parent.parent / "engines" / "typescript_engine" / "dist" / "index.js"
        if not scanner_path.exists():
            return False
        
        # 2. 檢查 Node.js 是否可用
        proc = await asyncio.create_subprocess_exec(
            "node", "--version",
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE
        )
        await proc.communicate()
        return proc.returncode == 0
    
    async def scan(
        self,
        targets: List[str],
        options: Dict[str, Any]
    ) -> Dict[str, Any]:
        """執行 TypeScript 引擎掃描"""
        all_assets = []
        
        for target in targets:
            # 1. 調用 Node.js 掃描器（異步子進程）
            proc = await asyncio.create_subprocess_exec(
                "node",
                str(self.ts_scanner_path),
                target,
                "--max-depth", str(options.get("max_depth", 3)),
                "--timeout", str(options.get("timeout", 10)),
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.PIPE
            )
            
            # 2. 等待執行完成（帶超時）
            stdout, stderr = await asyncio.wait_for(
                proc.communicate(),
                timeout=options.get("timeout", 10) + 10
            )
            
            # 3. 解析 JSON 輸出（健壯的解析器）
            ts_result = self._robust_parse_json(stdout.decode('utf-8'))
            
            # 4. 提取資產
            for asset_data in ts_result.get("assets", []):
                asset = Asset(
                    asset_id=asset_data.get("asset_id"),
                    type=asset_data.get("type", "url"),
                    value=asset_data.get("value", ""),
                    parameters=asset_data.get("parameters", []),
                    has_form=asset_data.get("has_form", False)
                )
                all_assets.append(asset)
        
        return {
            "assets": all_assets,
            "metadata": {
                "engine": "typescript",
                "targets_scanned": len(targets),
                "assets_found": len(all_assets)
            },
            "error": None
        }
```

**通信方式**:
- ✅ **異步子進程** (`asyncio.create_subprocess_exec`)
- ✅ **JSON 通信協議** (stdout/stderr)
- ✅ **健壯的解析器**（應對 console.log 污染）
- ✅ **超時保護**（避免掛起）

#### 2.3 Rust Adapter - FFI Bridge + 線程池

**位置**: `services/scan/coordinators/engines/rust_adapter.py`

```python
class RustAdapter(BaseScannerAdapter):
    """Rust 引擎適配器 - FFI Bridge + 線程池"""
    
    async def is_available(self) -> bool:
        """檢查 Rust 引擎是否可用"""
        try:
            # 嘗試導入 Rust FFI Bridge
            from services.scan.engines.rust_engine.python_bridge import rust_info_gatherer
            self.rust_scanner = rust_info_gatherer
            return True
        except ImportError as e:
            self.logger.warning(f"Rust Bridge 加載失敗: {e}")
            return False
    
    async def scan(
        self,
        targets: List[str],
        options: Dict[str, Any]
    ) -> Dict[str, Any]:
        """執行 Rust 引擎掃描"""
        loop = asyncio.get_event_loop()
        
        # ✅ 關鍵架構: 線程池包裝同步 FFI 調用
        with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
            
            def _sync_scan_all_targets():
                all_rust_assets = []
                
                for target in targets:
                    # 調用 Rust FFI
                    result = self.rust_scanner.scan_target(target, {
                        "mode": options.get("mode", "fast"),
                        "timeout": options.get("timeout", 10)
                    })
                    
                    # 提取資產
                    if result and result.get("success"):
                        raw_assets = result["results"].get("assets", [])
                        all_rust_assets.extend(raw_assets)
                
                return all_rust_assets
            
            # 在線程池中執行，釋放 GIL
            raw_assets = await loop.run_in_executor(pool, _sync_scan_all_targets)
        
        # 標準化資產
        standardized_assets = self._standardize_assets(raw_assets, "rust")
        
        return {
            "assets": standardized_assets,
            "metadata": {
                "engine": "rust",
                "targets_scanned": len(targets),
                "assets_found": len(standardized_assets)
            },
            "error": None
        }
```

**通信方式**:
- ✅ **FFI (Foreign Function Interface)** - 直接調用 Rust 函數
- ✅ **線程池包裝** - 避免阻塞 asyncio 事件循環
- ✅ **GIL 釋放** - `run_in_executor` 在後台線程執行

**關鍵技術點**:
```
Python Async Event Loop
   ↓
run_in_executor() → ThreadPoolExecutor
   ↓
Rust FFI Bridge (同步調用)
   ↓
Rust Scanner (編譯的動態庫)
```

```python
class GoAdapter(BaseScannerAdapter):
    """Go 引擎適配器 - 子進程調用"""
    
    async def is_available(self) -> bool:
        """檢查 Go 引擎是否可用"""
        # 檢查編譯後的 Go 二進制文件
        scanner_path = Path(__file__).parent.parent / "engines" / "go_engine" / "scanner"
        return scanner_path.exists()
    
    async def scan(
        self,
        targets: List[str],
        options: Dict[str, Any]
    ) -> Dict[str, Any]:
        """執行 Go 引擎掃描"""
        all_assets = []
        
        for target in targets:
            # 1. 調用 Go 二進制文件
            proc = await asyncio.create_subprocess_exec(
                str(self.go_scanner_path),
                "--url", target,
                "--timeout", str(options.get("timeout", 10)),
                "--concurrency", str(options.get("concurrency", 5)),
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.PIPE
            )
            
            # 2. 等待執行完成
            stdout, stderr = await asyncio.wait_for(
                proc.communicate(),
                timeout=options.get("timeout", 10) + 10
            )
            
            # 3. 解析 JSON 輸出
            go_result = json.loads(stdout.decode('utf-8'))
            
            # 4. 提取資產
            for asset_data in go_result.get("assets", []):
                asset = Asset(**asset_data)
                all_assets.append(asset)
        
        return {
            "assets": all_assets,
            "metadata": {
                "engine": "go",
                "targets_scanned": len(targets)
            },
            "error": None
        }
```

**通信方式**:
- ✅ **異步子進程** (`asyncio.create_subprocess_exec`)
- ✅ **JSON 通信協議** (stdout)
- ✅ **命令行參數傳遞** (`--url`, `--timeout`, `--concurrency`)

### 3. 協調器統一編排

**位置**: `services/scan/coordinators/multi_engine_coordinator.py`

```python
class MultiEngineCoordinator:
    """多引擎協調器 - 使用適配器模式"""
    
    def __init__(self, config: Optional[Dict[str, Any]] = None):
        self.config = config or {}
        self.logger = get_logger("MultiEngineCoordinator")
        
        # ✅ 適配器模式：初始化四引擎適配器
        from .engines import PythonAdapter, TypeScriptAdapter, RustAdapter, GoAdapter
        
        self.adapters = {
            "python": PythonAdapter(),
            "typescript": TypeScriptAdapter(),
            "rust": RustAdapter(),
            "go": GoAdapter(),
        }
        
        # 引擎可用性
        self.available_engines: Set[EngineType] = set()
    
    async def initialize(self) -> None:
        """初始化並檢查所有引擎的可用性"""
        self.logger.info("🔍 開始檢查引擎可用性...")
        
        for engine_name, adapter in self.adapters.items():
            try:
                is_available = await adapter.is_available()
                if is_available:
                    self.available_engines.add(EngineType(engine_name))
                    self.logger.info(f"  ✅ {engine_name.upper()} 引擎可用")
                else:
                    self.logger.warning(f"  ❌ {engine_name.upper()} 引擎不可用")
            except Exception as e:
                self.logger.error(f"  ❌ {engine_name.upper()} 引擎檢查失敗: {e}")
        
        if not self.available_engines:
            self.logger.error("⚠️  警告：沒有任何引擎可用！")
    
    async def execute_phase1(
        self,
        targets: List[str],
        options: Dict[str, Any],
        selected_engines: Optional[List[EngineType]] = None
    ) -> Phase1CompletedPayload:
        """執行 Phase 1 多引擎掃描"""
        
        # 確定要使用的引擎
        if selected_engines is None:
            selected_engines = list(self.available_engines)
        
        # ✅ 並行執行所有引擎
        tasks = []
        for engine_type in selected_engines:
            if engine_type in self.available_engines:
                adapter = self.adapters[engine_type.value]
                task = adapter.scan(targets, options)
                tasks.append((engine_type, task))
        
        # 等待所有引擎完成
        results = await asyncio.gather(
            *[task for _, task in tasks],
            return_exceptions=True
        )
        
        # 聚合結果
        all_assets = []
        for (engine_type, _), result in zip(tasks, results):
            if isinstance(result, Exception):
                self.logger.error(f"{engine_type.value} 引擎錯誤: {result}")
                continue
            
            # 提取資產
            assets = result.get("assets", [])
            all_assets.extend(assets)
        
        # 去重
        unique_assets = self._deduplicate_assets(all_assets)
        
        # 構建 Phase1CompletedPayload
        return Phase1CompletedPayload(
            scan_id=options.get("scan_id", "unknown"),
            assets=unique_assets,
            summary=Summary(
                urls_found=len([a for a in unique_assets if a.type == "url"]),
                forms_found=len([a for a in unique_assets if a.has_form]),
                scan_duration_seconds=time.time() - start_time
            ),
            fingerprints=Fingerprints(...)
        )
```

**核心設計**:
1. ✅ **依賴注入**: Coordinator 不依賴具體引擎實現
2. ✅ **並行執行**: `asyncio.gather()` 並行執行多引擎
3. ✅ **錯誤隔離**: 單引擎失敗不影響整體
4. ✅ **統一聚合**: 去重並生成標準化 Payload

## 🎯 關鍵技術總結

### 1. 統一接口 (Interface Segregation)

```python
# 所有適配器實現相同接口
class BaseScannerAdapter(ABC):
    async def is_available(self) -> bool
    async def scan(targets, options) -> Dict[str, Any]
    def _standardize_assets(raw_assets, engine_name) -> List[Asset]
```

### 2. 跨語言通信方式

| 引擎 | 通信方式 | 技術細節 |
|------|----------|----------|
| **Python** | 原生調用 | `from ... import ScanOrchestrator`<br>直接調用 Python 對象 |
| **TypeScript** | 子進程 | `asyncio.create_subprocess_exec("node", ...)`<br>JSON stdout/stderr |
| **Rust** | FFI Bridge | `from python_bridge import rust_scanner`<br>線程池 + `run_in_executor` |
| **Go** | 子進程 | `asyncio.create_subprocess_exec("./scanner", ...)`<br>JSON stdout |

### 3. 異步協調模式

```python
# 並行執行所有引擎
tasks = [
    python_adapter.scan(targets, options),
    typescript_adapter.scan(targets, options),
    rust_adapter.scan(targets, options),
    go_adapter.scan(targets, options)
]

# 等待所有完成，錯誤不影響其他引擎
results = await asyncio.gather(*tasks, return_exceptions=True)
```

## ✅ 如何應用到 Features 集成

### 方案：創建 FeaturesInvoker（參考掃描模組）

```python
# services/core/aiva_core/plugins/features_invoker.py

from abc import ABC, abstractmethod
from typing import List, Dict, Any

class BaseFeatureAdapter(ABC):
    """Features 適配器基類（參考 BaseScannerAdapter）"""
    
    @abstractmethod
    async def is_available(self) -> bool:
        """檢查 Feature 是否可用"""
        pass
    
    @abstractmethod
    async def execute(
        self,
        parameters: Dict[str, Any]
    ) -> Dict[str, Any]:
        """執行 Feature 任務
        
        Returns:
            {
                "result": {...},      # 標準化結果
                "metadata": {...},    # 執行元數據
                "error": None         # 錯誤訊息
            }
        """
        pass


class PythonFeatureAdapter(BaseFeatureAdapter):
    """Python Features 適配器"""
    
    async def is_available(self) -> bool:
        try:
            # 嘗試導入 Python Feature
            from services.features.func_xss.xss_detector import XSSDetector
            return True
        except ImportError:
            return False
    
    async def execute(self, parameters: Dict[str, Any]) -> Dict[str, Any]:
        # 原生 Python 調用
        from services.features.func_xss.xss_detector import XSSDetector
        
        detector = XSSDetector()
        result = await detector.scan(
            target=parameters["target"],
            options=parameters.get("options", {})
        )
        
        return {
            "result": result,
            "metadata": {"engine": "python", "feature": "xss"},
            "error": None
        }


class RustFeatureAdapter(BaseFeatureAdapter):
    """Rust Features 適配器（SSRF）"""
    
    async def is_available(self) -> bool:
        # 檢查 Rust 二進制文件
        scanner_path = Path("services/features/function_ssrf_rust/target/release/ssrf_scanner")
        return scanner_path.exists()
    
    async def execute(self, parameters: Dict[str, Any]) -> Dict[str, Any]:
        # 子進程調用
        proc = await asyncio.create_subprocess_exec(
            str(self.scanner_path),
            "--target", parameters["target"],
            "--timeout", str(parameters.get("timeout", 10)),
            stdout=asyncio.subprocess.PIPE
        )
        
        stdout, _ = await proc.communicate()
        result = json.loads(stdout.decode())
        
        return {
            "result": result,
            "metadata": {"engine": "rust", "feature": "ssrf"},
            "error": None
        }


class GoFeatureAdapter(BaseFeatureAdapter):
    """Go Features 適配器（認證掃描）"""
    
    async def is_available(self) -> bool:
        scanner_path = Path("services/features/function_authn_go/worker")
        return scanner_path.exists()
    
    async def execute(self, parameters: Dict[str, Any]) -> Dict[str, Any]:
        # 子進程調用
        proc = await asyncio.create_subprocess_exec(
            str(self.scanner_path),
            "--url", parameters["target"],
            stdout=asyncio.subprocess.PIPE
        )
        
        stdout, _ = await proc.communicate()
        result = json.loads(stdout.decode())
        
        return {
            "result": result,
            "metadata": {"engine": "go", "feature": "authn"},
            "error": None
        }


class FeaturesInvoker:
    """Features 調用器（參考 MultiEngineCoordinator）"""
    
    def __init__(self):
        self.adapters = {
            "xss": PythonFeatureAdapter(),
            "sqli": PythonFeatureAdapter(),
            "ssrf": RustFeatureAdapter(),
            "authn": GoFeatureAdapter(),
        }
    
    async def initialize(self) -> None:
        """初始化並檢查所有 Features 可用性"""
        for feature_name, adapter in self.adapters.items():
            is_available = await adapter.is_available()
            if is_available:
                logger.info(f"  ✅ {feature_name.upper()} Feature 可用")
            else:
                logger.warning(f"  ❌ {feature_name.upper()} Feature 不可用")
    
    async def invoke_feature(
        self,
        feature_name: str,
        parameters: Dict[str, Any]
    ) -> Dict[str, Any]:
        """統一調用 Features"""
        
        adapter = self.adapters.get(feature_name)
        if adapter is None:
            raise ValueError(f"Unknown feature: {feature_name}")
        
        # 檢查可用性
        if not await adapter.is_available():
            return {
                "result": None,
                "metadata": {"feature": feature_name},
                "error": f"Feature {feature_name} not available"
            }
        
        # 執行 Feature
        return await adapter.execute(parameters)
```
