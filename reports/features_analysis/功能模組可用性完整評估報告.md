### 6.2 批量檢測範例

```python
import asyncio
from services.features import function_xss, function_sqli, function_ssrf

async def batch_scan(target_url):
    """批量執行多種漏洞掃描"""
    
    # 並行執行多個掃描
    tasks = [
        function_xss.XSSCommandHandler().handle_command(
            AICommand(
                command_type=CommandType.FEATURE_XSS_TEST,
                payload={"target_url": target_url}
            )
        ),
        function_sqli.SQLiCommandHandler().handle_command(
            AICommand(
                command_type=CommandType.FEATURE_SQLI_TEST,
                payload={"target_url": target_url}
            )
        ),
        function_ssrf.SSRFCommandHandler().handle_command(
            AICommand(
                command_type=CommandType.FEATURE_SSRF_TEST,
                payload={"target_url": target_url}
            )
        )
    ]
    
    results = await asyncio.gather(*tasks, return_exceptions=True)
    
    return {
        "xss": results[0],
        "sqli": results[1],
        "ssrf": results[2]
    }

# 使用
results = await batch_scan("https://example.com")
```

### 6.3 錯誤處理最佳實踐

```python
from services.aiva_common.exceptions import CommandExecutionError

async def safe_scan(target_url):
    """安全的掃描執行,包含錯誤處理"""
    
    handler = XSSCommandHandler()
    
    try:
        command = AICommand(
            command_type=CommandType.FEATURE_XSS_TEST,
            payload={"target_url": target_url}
        )
        
        result = await handler.handle_command(command)
        
        if not result.success:
            print(f"⚠️ 掃描失敗: {result.error_message}")
            return None
            
        return result.data
        
    except CommandExecutionError as e:
        print(f"❌ 命令執行錯誤: {e}")
        return None
    except Exception as e:
        print(f"❌ 未預期錯誤: {e}")
        return None
```
