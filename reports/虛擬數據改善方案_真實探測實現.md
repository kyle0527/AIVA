import asyncio

class NetworkScanner:
    def __init__(self):

        self.nma = nmap.PortScannerAsync()
    
    async def _detect_service_version_real(self, host: str, port: int) -> Optional[Dict[str, str]]:
        """真實的服務版本檢測（使用 nmap -sV）"""
        try:
            # 使用 nmap 的服務版本檢測
            self.nm.scan(host, str(port), arguments='-sV')

                        "service": port_info.get('name', 'unknown'),

                        "cpe": port_info.get('cpe', ''),

                    }
            return None
            
        except Exception as e:

            return None
    
    async def _detect_os_real(self, host: str) -> Dict[str, Any]:
        """真實的 OS 檢測（使用 nmap -O）"""
        try:
            # 使用 nmap 的 OS 檢測（需要 root 權限）
            self.nm.scan(host, arguments='-O')
            
            if host in self.nm.all_hosts():
                os_info = {}
                
                # 獲取 osmatch 信息
                if 'osmatch' in self.nm[host]:
                    osmatch = self.nm[host]['osmatch']
                    if osmatch:
                        best_match = osmatch[0]  # 最匹配的結果
                        os_info['os_name'] = best_match.get('name', 'Unknown')
                        os_info['accuracy'] = best_match.get('accuracy', 0)
                        
                        # 獲取 osclass 詳細信息
                        if 'osclass' in best_match and best_match['osclass']:
                            osclass = best_match['osclass'][0]
                            os_info['os_family'] = osclass.get('osfamily', '')
                            os_info['os_gen'] = osclass.get('osgen', '')
                            os_info['vendor'] = osclass.get('vendor', '')
                            os_info['type'] = osclass.get('type', '')
                
                # 獲取 TCP/IP 指紋信息
                if 'portused' in self.nm[host]:
                    os_info['portused'] = self.nm[host]['portused']
                
                return os_info
            
            return {"os_guess": "Unknown", "accuracy": 0}
            
        except nmap.PortScannerError as e:
            logger.warning(f"OS 檢測失敗（可能需要 root 權限）: {e}")
            # 降級到簡單檢測
            return await self._detect_os_simple(host)
        except Exception as e:
            logger.error(f"OS 檢測異常: {e}")
            return {"os_guess": "Unknown", "accuracy": 0}
    
    async def _detect_os_simple(self, host: str) -> Dict[str, Any]:
        """簡單的 OS 檢測（無需 root）"""
        os_info = {"os_guess": "Unknown", "accuracy": 0, "method": "heuristic"}
        
        try:
            # 檢測常見 OS 特徵端口
            checks = await asyncio.gather(
                self._check_port(host, 135),   # Windows RPC
                self._check_port(host, 3389),  # Windows RDP
                self._check_port(host, 445),   # Windows SMB
                self._check_port(host, 22),    # Linux/Unix SSH
                self._check_port(host, 548),   # macOS AFP
                return_exceptions=True
            )
            
            windows_ports = sum(1 for i in [0, 1, 2] if checks[i] == "open")
            linux_ports = 1 if checks[3] == "open" else 0
            macos_ports = 1 if checks[4] == "open" else 0
            
            if windows_ports >= 2:
                os_info["os_guess"] = "Windows"
                os_info["accuracy"] = min(50 + windows_ports * 15, 95)
            elif macos_ports > 0:
                os_info["os_guess"] = "macOS"
                os_info["accuracy"] = 60
            elif linux_ports > 0:
                os_info["os_guess"] = "Linux/Unix"
                os_info["accuracy"] = 55
            
            return os_info
            
        except Exception as e:
            logger.error(f"簡單 OS 檢測失敗: {e}")
            return os_info
    
    async def _ping_real(self, host: str) -> Dict[str, Any]:
        """真實的 ICMP Ping（使用 nmap -sn）"""
        try:
            # 使用 nmap 的 ping scan
            self.nm.scan(host, arguments='-sn')  # -sn = ping scan only
            
            if host in self.nm.all_hosts():
                status = self.nm[host]['status']
                return {
                    "state": status['state'],      # up/down
                    "reason": status['reason'],    # echo-reply/no-response
                    "ping_success": status['state'] == 'up',
                    "method": "ICMP"
                }
            
            return {
                "state": "down",
                "reason": "no-response",
                "ping_success": False,
                "method": "ICMP"
            }
            
        except Exception as e:
            logger.warning(f"ICMP Ping 失敗，降級到 TCP Ping: {e}")
            # 降級到 TCP 連接測試
            return await self._ping_tcp(host)
    
    async def _ping_tcp(self, host: str, ports: List[int] = [80, 443, 22]) -> Dict[str, Any]:
        """TCP Ping 替代方案（無需特殊權限）"""
        for port in ports:
            status = await self._check_port(host, port, timeout=2.0)
            if status in ["open", "closed"]:  # 有回應就表示主機存活
                return {
                    "state": "up",
                    "reason": f"tcp-response-port-{port}",
                    "ping_success": True,
                    "method": "TCP"
                }
        
        return {
            "state": "down",
            "reason": "no-tcp-response",
            "ping_success": False,
            "method": "TCP"
        }

#### 方案 B：原生 Banner Grabbing（輕量級）

如果不想依賴 nmap，可以實現原生的 Banner Grabbing：

```python
async def _grab_banner_native(self, host: str, port: int) -> Dict[str, Any]:
    """原生 Banner Grabbing（無需 nmap）"""
    try:
        reader, writer = await asyncio.wait_for(
            asyncio.open_connection(host, port),
            timeout=3.0
        )
        
        # 對於某些服務，需要先發送請求
        if port in [80, 8080, 8000]:
            # HTTP 服務
            writer.write(b"HEAD / HTTP/1.0\r\n\r\n")
        elif port == 21:
            # FTP 服務（會自動返回 banner）
            pass
        elif port == 25:
            # SMTP 服務（會自動返回 banner）
            pass
        else:
            # 其他服務嘗試讀取
            pass
        
        await writer.drain()
        
        # 讀取 banner
        banner = await asyncio.wait_for(
            reader.read(2048),
            timeout=2.0
        )
        
        writer.close()
        await writer.wait_closed()
        
        banner_str = banner.decode('utf-8', errors='ignore').strip()
        
        # 解析 banner 提取版本信息
        version_info = self._parse_banner(banner_str, port)
        return version_info
        
    except Exception as e:
        logger.debug(f"Banner grabbing 失敗 {host}:{port} - {e}")
        return None

def _parse_banner(self, banner: str, port: int) -> Dict[str, Any]:
    """解析 Banner 提取版本信息"""
    import re
    
    result = {
        "raw_banner": banner[:200],  # 限制長度
        "service": "unknown",
        "product": "",
        "version": ""
    }
    
    # HTTP/HTTPS 服務
    if port in [80, 443, 8080, 8000, 8443]:
        # 匹配 Server: Apache/2.4.41 (Ubuntu)
        match = re.search(r'Server:\s+([^\r\n]+)', banner, re.I)
        if match:
            server = match.group(1)
            result["service"] = "HTTP"
            result["product"] = server
            
            # 提取版本號
            version_match = re.search(r'[\d.]+', server)
            if version_match:
                result["version"] = version_match.group(0)
    
    # SSH 服務
    elif port == 22:
        # 匹配 SSH-2.0-OpenSSH_8.2p1 Ubuntu-4ubuntu0.1
        match = re.search(r'SSH-([\d.]+)-(.+)', banner)
        if match:
            result["service"] = "SSH"
            result["product"] = match.group(2).split()[0]
            result["version"] = match.group(1)
    
    # FTP 服務
    elif port == 21:
        # 匹配 220 ProFTPD 1.3.5 Server
        match = re.search(r'220\s+(.+?)\s+([\d.]+)', banner)
        if match:
            result["service"] = "FTP"
            result["product"] = match.group(1)
            result["version"] = match.group(2)
    
    # MySQL
    elif port == 3306:
        if b'mysql' in banner.lower():
            result["service"] = "MySQL"
            # MySQL banner 是二進制協議，需要特殊解析
            version_match = re.search(r'([\d.]+)-', banner)
            if version_match:
                result["version"] = version_match.group(1)
    
    return result
```

async def detect_service_version(self, host: str, port: int) -> Dict[str, Any]:
    """智能服務版本檢測（三層策略）"""
    
    # 第一層：嘗試使用 nmap（最準確）
    if self.nmap_available:
        result = await self._detect_service_version_nmap(host, port)
        if result:
            result['detection_method'] = 'nmap'
            return result
    
    # 第二層：原生 Banner Grabbing（中等準確度）
    result = await self._grab_banner_native(host, port)
    if result:
        result['detection_method'] = 'banner_grabbing'
        return result
    
    # 第三層：基於端口的推測（最低準確度）
    result = self._guess_by_port(port)
    result['detection_method'] = 'heuristic'
    result['confidence'] = 'low'
    return result

def _guess_by_port(self, port: int) -> Dict[str, Any]:
    """基於端口的服務推測（最後手段）"""
    # 使用 IANA 端口數據庫
    service_map = {
        22: {"service": "SSH", "product": "OpenSSH", "version": ""},
        80: {"service": "HTTP", "product": "Apache", "version": ""},
        443: {"service": "HTTPS", "product": "nginx", "version": ""},
        3306: {"service": "MySQL", "product": "MySQL", "version": ""},
        # ... 更多端口映射
    }
    
    return service_map.get(port, {
        "service": f"port-{port}",
        "product": "unknown",
        "version": ""
    })

import sqlite3
import json
from datetime import datetime

class RealExperienceManager:
    """真實的經驗管理器（使用 SQLite）"""
    
    def __init__(self, db_path: str = "data/experience.db"):
        self.db_path = db_path
        self._init_database()
    
    def _init_database(self):
        """初始化數據庫"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS experiences (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                timestamp TEXT NOT NULL,
                situation TEXT NOT NULL,
                action TEXT NOT NULL,
                outcome TEXT NOT NULL,
                success INTEGER NOT NULL,
                execution_time REAL,
                metadata TEXT
            )
        ''')
        
        cursor.execute('''
            CREATE INDEX IF NOT EXISTS idx_timestamp 
            ON experiences(timestamp DESC)
        ''')
        
        cursor.execute('''
            CREATE INDEX IF NOT EXISTS idx_success 
            ON experiences(success)
        ''')
        
        conn.commit()
        conn.close()
    
    def add_sample(self, sample: Dict[str, Any]) -> None:
        """添加經驗樣本"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute('''
            INSERT INTO experiences 
            (timestamp, situation, action, outcome, success, execution_time, metadata)
            VALUES (?, ?, ?, ?, ?, ?, ?)
        ''', (
            sample.get('timestamp', datetime.now().isoformat()),
            json.dumps(sample.get('situation', {})),
            json.dumps(sample.get('action', {})),
            json.dumps(sample.get('outcome', {})),
            1 if sample.get('success') else 0,
            sample.get('execution_time', 0.0),
            json.dumps(sample.get('metadata', {}))
        ))
        
        conn.commit()
        conn.close()
    
    def get_statistics(self) -> Dict[str, Any]:
        """獲取統計信息"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        # 總樣本數
        cursor.execute('SELECT COUNT(*) FROM experiences')
        total = cursor.fetchone()[0]
        
        # 成功率
        cursor.execute('SELECT COUNT(*) FROM experiences WHERE success = 1')
        success_count = cursor.fetchone()[0]
        
        # 平均執行時間
        cursor.execute('SELECT AVG(execution_time) FROM experiences')
        avg_time = cursor.fetchone()[0] or 0.0
        
        # 最近 24 小時統計
        cursor.execute('''
            SELECT COUNT(*) FROM experiences 
            WHERE timestamp > datetime('now', '-1 day')
        ''')
        recent_24h = cursor.fetchone()[0]
        
        conn.close()
        
        return {
            "total_samples": total,
            "success_count": success_count,
            "success_rate": success_count / total if total > 0 else 0.0,
            "avg_execution_time": avg_time,
            "recent_24h": recent_24h
        }
    
    def query_similar(self, situation: Dict[str, Any], limit: int = 10) -> List[Dict]:
        """查詢相似情況的歷史經驗"""
        # 簡化版：根據關鍵字匹配
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        # 提取關鍵特徵
        target_type = situation.get('target_type', '')
        
        cursor.execute('''
            SELECT situation, action, outcome, success, execution_time
            FROM experiences
            WHERE situation LIKE ?
            ORDER BY timestamp DESC
            LIMIT ?
        ''', (f'%{target_type}%', limit))
        
        results = []
        for row in cursor.fetchall():
            results.append({
                'situation': json.loads(row[0]),
                'action': json.loads(row[1]),
                'outcome': json.loads(row[2]),
                'success': bool(row[3]),
                'execution_time': row[4]
            })
        
        conn.close()
        return results

#### AI 風險評估模型（簡化版）

```python
from sklearn.ensemble import RandomForestClassifier
import joblib

class RiskAssessmentModel:
    """AI 風險評估模型（使用機器學習）"""
    
    def __init__(self, model_path: str = "models/risk_model.pkl"):
        self.model_path = model_path
        self.model = self._load_or_create_model()
        self.feature_names = [
            'target_type_production',
            'authorized',
            'scan_intensity',
            'target_sensitivity',
            'time_of_day'
        ]
    
    def _load_or_create_model(self):
        """加載或創建模型"""
        try:
            return joblib.load(self.model_path)
        except FileNotFoundError:
            # 創建新模型
            model = RandomForestClassifier(n_estimators=100, random_state=42)
            return model
    
    def extract_features(self, situation: Dict, constraints: Dict) -> List[float]:
        """提取特徵向量"""
        features = [
            1.0 if situation.get("target_type") == "production" else 0.0,
            1.0 if constraints.get("authorized") else 0.0,
            float(situation.get("scan_intensity", 0)),
            float(situation.get("target_sensitivity", 0)),
            float(datetime.now().hour / 24.0)  # 時間因素
        ]
        return features
    
    def assess_risk(self, situation: Dict, constraints: Dict) -> Dict[str, Any]:
        """評估風險（使用 AI 模型）"""
        features = self.extract_features(situation, constraints)
        
        # 如果模型已訓練，使用預測
        if hasattr(self.model, 'n_features_in_'):
            risk_prob = self.model.predict_proba([features])[0][1]
            risk_score = int(risk_prob * 10)
        else:
            # 模型未訓練，使用啟發式規則
            risk_score = self._heuristic_risk(situation, constraints)
        
        return {
            "risk_score": risk_score,
            "risk_level": "high" if risk_score >= 7 else "medium" if risk_score >= 4 else "low",
            "features": dict(zip(self.feature_names, features)),
            "model_trained": hasattr(self.model, 'n_features_in_')
        }
    
    def _heuristic_risk(self, situation: Dict, constraints: Dict) -> int:
        """啟發式風險評估（模型未訓練時的回退）"""
        risk = 0
        
        if situation.get("target_type") == "production":
            risk += 3
        if not constraints.get("authorized"):
            risk += 5
        if situation.get("scan_intensity", 0) > 0.7:
            risk += 2
        
        return min(risk, 10)
    
    def train(self, X, y):
        """訓練模型"""
        self.model.fit(X, y)
        joblib.dump(self.model, self.model_path)
```

## 五、配置管理

### config/scanner_config.yaml

```yaml
network_scanner:
  # 掃描模式：real（真實）/ simulated（模擬）
  mode: real
  
  # nmap 設置
  nmap:
    enabled: true
    timeout: 30
    sudo_required: false  # OS 檢測需要 root
  
  # Banner Grabbing 設置
  banner_grabbing:
    enabled: true
    timeout: 3
  
  # 降級策略
  fallback:
    use_heuristic: true  # 當真實檢測失敗時使用啟發式
    
ai_commander:
  experience_manager:
    type: sqlite  # sqlite / json / mock
    db_path: "data/experience.db"
  
  risk_assessment:
    type: ml  # ml / heuristic / hybrid
    model_path: "models/risk_model.pkl"
    auto_train: true
```

## 六、測試驗證

### 測試用例

```python
import pytest

class TestRealScanning:
    async def test_real_service_detection(self):
        """測試真實服務檢測"""
        scanner = NetworkScanner(use_real_detection=True)
        
        # 掃描本地 SSH
        result = await scanner._detect_service_version_real('localhost', 22)
        
        assert result is not None
        assert 'service' in result
        assert result['service'].lower() == 'ssh'
        assert 'version' in result
        assert result['version'] != ''  # 不是空的假數據
    
    async def test_os_detection(self):
        """測試 OS 檢測"""
        scanner = NetworkScanner(use_real_detection=True)
        
        result = await scanner._detect_os_real('scanme.nmap.org')
        
        assert result is not None
        assert 'os_name' in result
        assert result['os_name'] != 'Unknown'
    
    async def test_fallback_mechanism(self):
        """測試降級機制"""
        scanner = NetworkScanner(use_real_detection=False)
        
        # 應該降級到模擬模式
        result = await scanner.detect_service_version('example.com', 80)
        
        assert result is not None
        assert 'detection_method' in result
```
