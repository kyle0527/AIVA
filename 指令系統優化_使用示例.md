# 指令系統優化 - 使用示例

## 功能概述

本次優化為 AIVA 系統添加了以下功能：

1. **實時 UI 更新機制** - 命令執行過程中可以實時推送進度、狀態和中間結果到 UI
2. **網路搜索能力** - 整合多個搜索引擎和威脅情報源，支持 18 種搜索命令類型

---

## 1. 使用回調機制實現實時 UI 更新

### 基本用法

```python
from services.aiva_common.schemas.commands import AICommand, CommandType
from services.aiva_common.command_center import get_command_center
from services.core.aiva_core.ui_panel.command_callback import UICommandCallback

# 1. 創建 UI 回調處理器
ui_callback = UICommandCallback(
    use_websocket=True,      # 使用 WebSocket 推送
    use_rich_console=True    # 使用 Rich Console 輸出
)

# 2. 創建命令並啟用回調
command = AICommand(
    command_id="scan_001",
    command_type=CommandType.SCAN_PHASE0,
    target_module="scan",
    payload={
        "scan_id": "scan_001",
        "targets": ["https://example.com"]
    },
    enable_callbacks=True,  # ✨ 啟用回調
    ui_update_interval=1.0  # UI 更新間隔（秒）
)

# 3. 設置回調處理器
command.set_callback(ui_callback)

# 4. 執行命令（會實時推送更新）
command_center = get_command_center()
result = await command_center.execute(command)
```

### 自定義回調處理器

```python
from services.aiva_common.schemas.commands import CommandCallback, CommandStatus

class MyCustomCallback:
    """自定義回調處理器"""
    
    async def on_progress(
        self, 
        command_id: str, 
        progress: float, 
        message: str,
        metadata: dict = None
    ):
        """進度更新"""
        print(f"[{command_id}] {progress:.1%} - {message}")
        
        # 發送到自定義的消息系統
        await my_message_queue.send({
            "type": "progress",
            "command_id": command_id,
            "progress": progress,
            "message": message
        })
    
    async def on_status_change(
        self,
        command_id: str,
        old_status: CommandStatus,
        new_status: CommandStatus,
        metadata: dict = None
    ):
        """狀態變更"""
        print(f"[{command_id}] {old_status.value} → {new_status.value}")
        
        # 更新數據庫
        await db.update_command_status(command_id, new_status.value)
    
    async def on_partial_result(
        self,
        command_id: str,
        result_type: str,
        data: any,
        metadata: dict = None
    ):
        """中間結果"""
        print(f"[{command_id}] 發現 {result_type}: {data}")
        
        # 存儲中間結果
        await db.save_partial_result(command_id, result_type, data)

# 使用自定義回調
custom_callback = MyCustomCallback()
command.set_callback(custom_callback)
```

## 3. AI 自主搜索與決策

### 示例：AI 發現未知漏洞後自動搜索

```python
class AIVulnerabilityAnalyzer:
    """AI 漏洞分析器（支持自主搜索）"""
    
    def __init__(self):
        self.command_center = get_command_center()
        self.ui_callback = UICommandCallback()
    
    async def analyze_vulnerability(self, vuln_data: dict):
        """分析漏洞並自動搜索相關資訊"""
        
        vuln_type = vuln_data.get("type", "unknown")
        
        # 1. 判斷是否需要搜索背景資訊
        if self._is_unknown_vulnerability(vuln_type):
            print(f"🤖 AI: 發現未知漏洞類型 '{vuln_type}'，開始搜索...")
            
            # 2. 並行搜索多個來源
            search_tasks = [
                self._search_google(vuln_type),
                self._search_github(vuln_type),
                self._search_cve(vuln_type),
                self._search_exploitdb(vuln_type)
            ]
            
            results = await asyncio.gather(*search_tasks, return_exceptions=True)
            
            # 3. 分析搜索結果
            knowledge = self._analyze_search_results(results)
            
            # 4. 根據搜索結果制定攻擊計劃
            attack_plan = self._create_attack_plan(vuln_data, knowledge)
            
            print(f"🤖 AI: 已制定攻擊計劃，包含 {len(attack_plan)} 個步驟")
            
            return attack_plan
        
        else:
            # 已知漏洞，直接使用內部知識
            return self._use_internal_knowledge(vuln_type)
    
    async def _search_google(self, query: str):
        """Google 搜索"""
        command = AICommand(
            command_id=f"auto_search_google_{int(time.time())}",
            command_type=CommandType.SEARCH_GOOGLE,
            target_module="search",
            payload={
                "query": f"{query} vulnerability exploitation",
                "max_results": 5
            },
            enable_callbacks=True
        )
        command.set_callback(self.ui_callback)
        
        return await self.command_center.execute(command)
    
    async def _search_github(self, query: str):
        """GitHub 搜索 POC"""
        command = AICommand(
            command_id=f"auto_search_github_{int(time.time())}",
            command_type=CommandType.SEARCH_GITHUB,
            target_module="search",
            payload={
                "query": f"{query} POC exploit",
                "search_type": "code",
                "max_results": 10
            },
            enable_callbacks=True
        )
        command.set_callback(self.ui_callback)
        
        return await self.command_center.execute(command)
    
    def _analyze_search_results(self, results: list) -> dict:
        """分析所有搜索結果，提取關鍵資訊"""
        knowledge = {
            "exploitation_methods": [],
            "poc_available": False,
            "severity_assessment": "unknown",
            "prevention_methods": []
        }
        
        # 分析 Google 搜索結果
        google_result = results[0]
        if google_result.success:
            # 提取關鍵詞、連結等
            pass
        
        # 分析 GitHub 搜索結果
        github_result = results[1]
        if github_result.success:
            if github_result.result.get("total_count", 0) > 0:
                knowledge["poc_available"] = True
        
        return knowledge
    
    def _create_attack_plan(self, vuln_data: dict, knowledge: dict) -> list:
        """根據漏洞數據和搜索知識創建攻擊計劃"""
        plan = []
        
        # 根據搜索到的知識制定計劃
        if knowledge["poc_available"]:
            plan.append({
                "step": 1,
                "action": "download_poc",
                "description": "從 GitHub 下載 POC"
            })
        
        plan.append({
            "step": 2,
            "action": "test_vulnerability",
            "description": "測試漏洞是否存在"
        })
        
        # ... 更多步驟
        
        return plan

# 使用
analyzer = AIVulnerabilityAnalyzer()

vuln_data = {
    "type": "未知的 XXE 變體",
    "target": "https://example.com/api/upload"
}

attack_plan = await analyzer.analyze_vulnerability(vuln_data)
```

## 4. WebSocket 實時通知

### 前端 JavaScript 客戶端

```javascript
// 連接 WebSocket
const ws = new WebSocket('ws://localhost:8000/ws/commands');

ws.onopen = () => {
    console.log('✅ WebSocket 已連接');
};

ws.onmessage = (event) => {
    const data = JSON.parse(event.data);
    
    switch (data.type) {
        case 'progress_update':
            updateProgressBar(data.command_id, data.progress);
            updateStatusText(data.command_id, data.message);
            break;
        
        case 'status_change':
            updateStatusBadge(data.command_id, data.status);
            break;
        
        case 'result':
            displayResult(data.command_id, data.result_type, data.data);
            break;
        
        case 'error':
            showError(data.command_id, data.error_message);
            break;
    }
};

// 更新進度條
function updateProgressBar(commandId, progress) {
    const progressBar = document.getElementById(`progress-${commandId}`);
    if (progressBar) {
        progressBar.style.width = `${progress * 100}%`;
        progressBar.textContent = `${(progress * 100).toFixed(1)}%`;
    }
}

// 更新狀態文本
function updateStatusText(commandId, message) {
    const statusText = document.getElementById(`status-${commandId}`);
    if (statusText) {
        statusText.textContent = message;
    }
}

// 顯示結果
function displayResult(commandId, resultType, data) {
    const resultsContainer = document.getElementById('results-container');
    
    const resultCard = document.createElement('div');
    resultCard.className = 'result-card';
    resultCard.innerHTML = `
        <h4>${resultType}</h4>
        <pre>${JSON.stringify(data, null, 2)}</pre>
        <small>Command ID: ${commandId}</small>
    `;
    
    resultsContainer.insertBefore(resultCard, resultsContainer.firstChild);
}
```

## 5. 完整示例：AI 自動化掃描與搜索

```python
async def automated_security_scan():
    """AI 自動化安全掃描（結合搜索和實時更新）"""
    
    # 初始化
    command_center = get_command_center()
    ui_callback = UICommandCallback(use_websocket=True, use_rich_console=True)
    
    target = "https://example.com"
    
    # 步驟 1: 執行 Phase 0 掃描
    print("\n🔍 步驟 1: 執行 Phase 0 掃描...")
    
    scan_command = AICommand(
        command_id="auto_scan_001",
        command_type=CommandType.SCAN_PHASE0,
        target_module="scan",
        payload={"targets": [target]},
        enable_callbacks=True
    )
    scan_command.set_callback(ui_callback)
    
    scan_result = await command_center.execute(scan_command)
    
    # 步驟 2: 分析掃描結果
    if scan_result.success:
        discovered_techs = scan_result.result.get("technologies", [])
        print(f"\n✅ 發現 {len(discovered_techs)} 種技術")
        
        # 步驟 3: 對每種技術搜索已知漏洞
        print("\n🔍 步驟 3: 搜索已知漏洞...")
        
        for tech in discovered_techs:
            tech_name = tech.get("name")
            tech_version = tech.get("version")
            
            # 搜索 CVE
            cve_command = AICommand(
                command_id=f"search_cve_{tech_name}",
                command_type=CommandType.SEARCH_CVE_DETAILS,
                target_module="search",
                payload={
                    "keyword": f"{tech_name} {tech_version}",
                    "max_results": 5
                },
                enable_callbacks=True
            )
            cve_command.set_callback(ui_callback)
            
            cve_result = await command_center.execute(cve_command)
            
            if cve_result.success:
                vulns = cve_result.result.get("vulnerabilities", [])
                
                if vulns:
                    print(f"\n⚠️  {tech_name} {tech_version} 發現 {len(vulns)} 個 CVE")
                    
                    # 對高危漏洞搜索 POC
                    for vuln in vulns:
                        if vuln.get("severity") in ["HIGH", "CRITICAL"]:
                            print(f"\n🔍 搜索 {vuln['id']} 的 POC...")
                            
                            poc_command = AICommand(
                                command_id=f"search_poc_{vuln['id']}",
                                command_type=CommandType.SEARCH_GITHUB,
                                target_module="search",
                                payload={
                                    "query": f"{vuln['id']} POC exploit",
                                    "search_type": "code",
                                    "max_results": 3
                                },
                                enable_callbacks=True
                            )
                            poc_command.set_callback(ui_callback)
                            
                            poc_result = await command_center.execute(poc_command)
                            
                            if poc_result.success and poc_result.result.get("total_count", 0) > 0:
                                print(f"✅ 找到 {vuln['id']} 的 POC！")
    
    print("\n✅ 自動化掃描完成！")

# 執行
asyncio.run(automated_security_scan())
```

## 總結

本次優化為 AIVA 系統帶來了：

1. **實時反饋** - 命令執行過程不再是黑盒，用戶可以實時看到進度
2. **智能搜索** - AI 可以自主搜索外部資源補充知識
3. **靈活擴展** - 易於添加新的搜索源和回調處理器
4. **完整控制** - AI Commander 保持對所有流程的完整掌控

現在 AI 可以像真正的安全專家一樣：
- 👀 實時觀察執行進度
- 🔍 自主搜索未知漏洞的資訊
- 🧠 根據搜索結果動態調整策略
- 📊 向用戶實時報告進展
