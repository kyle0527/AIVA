```python
# 整合點 1: 任務分解時查詢 RAG
async def _decompose_mission(self, target: str, intent: str) -> list[dict]:
    """查詢 RAG 獲取可用能力清單"""
    capabilities = await self.internal_loop_connector.query_self_awareness(
        query=f"針對 {intent} 任務，我有哪些可用能力？健康度如何？"
    )
    # 返回結果：
    # [
    #   {"name": "xss_scan", "health": 0.9, "description": "...", "params": {...}},
    #   {"name": "sqli_scan", "health": 0.85, "description": "...", "params": {...}}
    # ]
    return self._generate_tasks_from_capabilities(target, intent, capabilities)

# 整合點 2: 智能排序時參考健康度
def _calculate_task_priority(self, task: dict) -> int:
    """計算優先級，包含能力健康度"""
    score = 0
    
    # 原有權重
    score += self.type_weights.get(task['type'], 0)
    score += task.get('risk_level', 0) * 10
    
    # 新增：能力健康度權重
    health_score = self.capability_health_scores.get(task['capability_id'], 0.5)
    if health_score > 0.8:
        score -= 5  # 健康能力優先
    elif health_score < 0.3:
        score += 10  # 不健康能力延後
    
    return score
```

#### 3.2 外部CLI（待創建 🔲 → ✅）

**位置**: `services/core/aiva_core/internal_exploration/python_tools/external_module_cli_executor.py`

**功能**: 統一調用 Features/Scan 模組

```python
class ExternalModuleCLIExecutor:
    """外部模組 CLI 執行器"""
    
    def __init__(self):
        self.module_configs = {
            # ===== Features 模組 =====
            "xss": {
                "command": ["python", "-m", "services.features.function_xss"],
                "timeout": 30,
                "category": "features"
            },
            "sqli": {
                "command": ["python", "-m", "services.features.function_sqli"],
                "timeout": 30,
                "category": "features"
            },
            "ssrf": {
                "command": ["python", "-m", "services.features.function_ssrf"],
                "timeout": 30,
                "category": "features"
            },
            
            # ===== Scan 模組 =====
            "rust_scan": {
                "command": ["cargo", "run", "--release", 
                           "--manifest-path", "services/scan/rust_engine/Cargo.toml"],
                "timeout": 10,
                "category": "scan"
            },
            "go_scan": {
                "command": ["go", "run", "services/scan/go_engine/cmd/main.go"],
                "timeout": 15,
                "category": "scan"
            },
            "ts_scan": {
                "command": ["npm", "run", "cli", "--prefix", "services/scan/typescript_engine"],
                "timeout": 20,
                "category": "scan"
            }
        }
    
    async def execute_module(self, module: str, action: str, params: dict) -> dict:
        """執行外部模組並返回標準 JSON"""
        config = self.module_configs.get(module)
        if not config:
            return {"status": "failed", "error": f"Unknown module: {module}"}
        
        # 構建命令
        cmd = config["command"] + [
            "--action", action,
            "--target", params["target"]
        ]
        
        # subprocess 執行
        process = await asyncio.create_subprocess_exec(
            *cmd,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE
        )
        
        stdout, stderr = await asyncio.wait_for(
            process.communicate(),
            timeout=config["timeout"]
        )
        
        # 解析 JSON 輸出
        return json.loads(stdout.decode('utf-8'))
    
    async def execute_batch(self, tasks: list[dict]) -> list[dict]:
        """批量並發執行"""
        coroutines = [
            self.execute_module(
                module=task["module"],
                action=task.get("action", "scan"),
                params=task["params"]
            )
            for task in tasks
        ]
        return await asyncio.gather(*coroutines)
```
