# 需要在 BioNeuronDecisionController 中添加
async def make_decision(self, user_input: str):
    # 1. 查詢 RAG 獲取能力
    capabilities = await self.internal_loop.query_capabilities(user_input)
    
    # 2. 基於能力生成決策
    decision = await self.decision_agent.make_decision(
        context=user_input,
        available_capabilities=capabilities
    )
    
    # 3. 生成 SystemCommand
    command = self.generate_system_command(decision)
    
    # 4. 調度執行
    result = await self.command_executor.execute(command)
    return result
