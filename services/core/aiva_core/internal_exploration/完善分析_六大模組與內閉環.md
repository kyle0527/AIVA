async def query_my_capabilities(self, task_description: str) -> List[Capability]:
    """查詢內閉環數據 - 我有什麼能力能完成這個任務?"""
    
    # 1. 調用 RAG 引擎查詢
    from ..cognitive_core.rag_engine import rag_engine
    results = await rag_engine.query(
        query=task_description,
        collection_name="capability_metadata",
        top_k=5
    )
    
    # 2. 解析查詢結果
    capabilities = []
    for result in results:
        capability = Capability(
            name=result.metadata['name'],
            module=result.metadata['module'],
            usage=result.metadata['usage'],
            confidence=result.similarity_score
        )
        capabilities.append(capability)
    
    return capabilities

**現狀**:
- ✅ 知道流程: `bio → trainer → rl → models`
- ❌ 不知道 `rl_models.py` 有哪些類

**解決方案**: 新增結構分析功能

```python
# 在 capability_analyzer.py 添加

class CodeStructureExtractor(ast.NodeVisitor):
    """代碼結構提取器 - 補充現有 AST 分析"""
    
    def __init__(self):
        self.classes = []
        self.functions = []
        self.imports = []
    
    def visit_ClassDef(self, node: ast.ClassDef):
        """提取類定義"""
        class_info = {
            "name": node.name,
            "bases": [self._get_name(base) for base in node.bases],
            "methods": [],
            "line_start": node.lineno,
            "docstring": ast.get_docstring(node)
        }
        
        # 提取方法
        for item in node.body:
            if isinstance(item, ast.FunctionDef):
                class_info["methods"].append({
                    "name": item.name,
                    "params": [arg.arg for arg in item.args.args],
                    "line": item.lineno
                })
        
        self.classes.append(class_info)
        self.generic_visit(node)
    
    def visit_Import(self, node: ast.Import):
        """提取 import 語句"""
        for alias in node.names:
            self.imports.append({
                "type": "import",
                "module": alias.name,
                "alias": alias.asname
            })
        self.generic_visit(node)
    
    def visit_ImportFrom(self, node: ast.ImportFrom):
        """提取 from ... import 語句"""
        for alias in node.names:
            self.imports.append({
                "type": "from_import",
                "module": node.module,
                "name": alias.name,
                "alias": alias.asname
            })
        self.generic_visit(node)
```
