### AICommander 整合

#### 完整使用範例

```python
from services.core.aiva_core.external_learning import ExperienceManager
from services.integration.aiva_integration.unified_data_manager import (
    get_unified_data_manager,
)
import logging

logger = logging.getLogger(__name__)

class AICommander:
    """AI 指揮官 - 負責決策和攻擊執行"""
    
    def __init__(self):
        # 獲取整合模組管理器
        self.integration_manager = get_unified_data_manager()
        
        # 創建經驗管理器（啟用自動持久化）
        self.experience_manager = ExperienceManager(
            auto_persist=True,
            unified_manager=self.integration_manager,
        )
        
        logger.info("✅ AICommander 已連接整合模組")
    
    async def execute_attack(self, target: str, attack_type: str):
        """執行攻擊並記錄經驗"""
        
        # 步驟 1: 載入歷史高質量經驗
        historical_experiences = self.integration_manager.query_high_quality_experiences(
            attack_type=attack_type,
            min_score=0.8,
            limit=50,
        )
        logger.info(f"載入 {len(historical_experiences)} 個歷史經驗")
        
        # 步驟 2: 基於歷史經驗優化策略
        strategy = self._optimize_strategy(historical_experiences)
        logger.info(f"策略優化完成: {strategy}")
        
        # 步驟 3: 執行攻擊
        result = await self._execute_attack_plan(target, attack_type, strategy)
        
        # 步驟 4: 記錄經驗（自動保存到整合模組）
        self.experience_manager.push(
            state={"target": target, "ast": result.ast_graph},
            action={"type": attack_type, "params": result.params},
            next_state={"findings": result.findings},
            reward=result.success_rate,
            metadata={
                "plan_id": result.plan_id,
                "attack_type": attack_type,
                "execution_trace": result.trace,
            },
        )
        
        logger.info("✅ 經驗已自動保存到整合模組")
        
        return result
    
    def _optimize_strategy(self, experiences):
        """基於歷史經驗優化策略"""
        if not experiences:
            return self._get_default_strategy()
        
        # 分析成功模式
        successful_patterns = []
        for exp in experiences:
            if exp.overall_score >= 0.85:
                ast_graph = exp.get_ast_graph()
                if ast_graph:
                    successful_patterns.append(ast_graph.get("strategy"))
        
        # 選擇最常見的成功策略
        from collections import Counter
        strategy_counts = Counter(successful_patterns)
        most_common_strategy = strategy_counts.most_common(1)[0][0] if strategy_counts else None
        
        return most_common_strategy or self._get_default_strategy()
    
    def _get_default_strategy(self):
        """獲取預設策略"""
        return {"approach": "standard", "aggressiveness": 0.5}
    
    async def _execute_attack_plan(self, target, attack_type, strategy):
```

### RAGEngine 整合

#### 完整使用範例

```python
from services.core.aiva_core.rag_engine import RAGEngine
from services.integration.aiva_integration.unified_data_manager import (
    get_unified_data_manager,
)
import logging

logger = logging.getLogger(__name__)

class RAGEngine:
    """RAG 引擎 - 檢索增強生成"""
    
    def __init__(self):
        self.integration_manager = get_unified_data_manager()
        logger.info("✅ RAGEngine 已連接整合模組")
    
    async def enhance_prompt_with_experiences(
        self,
        attack_type: str,
        target_context: dict,
    ) -> str:
        """使用歷史經驗增強 Prompt"""
        
        # 步驟 1: 查詢相關經驗
        experiences = self.integration_manager.query_high_quality_experiences(
            attack_type=attack_type,
            min_score=0.85,
            limit=10,
        )
        
        logger.info(f"檢索到 {len(experiences)} 個高質量經驗")
        
        if not experiences:
            return self._build_basic_prompt(attack_type, target_context)
        
        # 步驟 2: 構建增強 Prompt
        prompt_parts = [
            "## Context: Based on successful historical attacks\n",
        ]
        
        for idx, exp in enumerate(experiences, 1):
            ast_graph = exp.get_ast_graph()
            execution_trace = exp.get_execution_trace()
            
            prompt_parts.append(
                f"### Experience {idx} (Score: {exp.overall_score:.2f})\n"
                f"- Attack Type: {exp.attack_type}\n"
                f"- Strategy: {ast_graph.get('strategy', 'N/A')}\n"
                f"- Success Rate: {execution_trace.get('success_rate', 'N/A')}\n"
                f"- Key Steps: {execution_trace.get('key_steps', [])}\n"
            )
        
        # 步驟 3: 添加當前任務
        prompt_parts.append(f"\n## Current Task\n")
        prompt_parts.append(f"Target: {target_context.get('target')}\n")
        prompt_parts.append(f"Objective: {target_context.get('objective')}\n")
        prompt_parts.append(f"\n## Instructions\n")
        prompt_parts.append(
            f"Based on the successful experiences above, generate an optimized attack plan.\n"
        )
        
        enhanced_prompt = "\n".join(prompt_parts)
        
        logger.info(f"✅ Prompt 已增強 (長度: {len(enhanced_prompt)} 字符)")
        
        return enhanced_prompt
    
    def _build_basic_prompt(self, attack_type: str, target_context: dict) -> str:
        """構建基本 Prompt（無歷史經驗）"""
        return f"Generate an attack plan for {attack_type} against {target_context.get('target')}"
    
    async def retrieve_similar_experiences(
        self,
        current_state: dict,
        attack_type: str,
        top_k: int = 5,
    ):
        """檢索相似經驗（用於 Few-shot Learning）"""
        
        # 查詢高質量經驗
        experiences = self.integration_manager.query_high_quality_experiences(
            attack_type=attack_type,
            min_score=0.8,
            limit=50,
        )
        
        # 計算相似度並排序
        similar_experiences = []
        for exp in experiences:
            similarity = self._calculate_similarity(current_state, exp)
            similar_experiences.append((exp, similarity))
        
        # 返回 Top-K
        similar_experiences.sort(key=lambda x: x[1], reverse=True)
        return [exp for exp, _ in similar_experiences[:top_k]]
    
    def _calculate_similarity(self, current_state: dict, experience) -> float:
        """計算狀態相似度"""
        # 簡化的相似度計算（實際應使用向量嵌入）
        score = 0.0
        
        # 比較目標類型
        if current_state.get("target_type") == experience.target_info.get("target_type"):
            score += 0.5
        
        # 比較攻擊類型
        if current_state.get("attack_type") == experience.attack_type:
            score += 0.5
        
        return score
```

#### 關鍵要點

1. **Prompt 增強**: 使用歷史經驗增強 LLM Prompt
2. **Few-shot Learning**: 檢索相似經驗用於 Few-shot
3. **質量過濾**: 只使用高分經驗（min_score >= 0.85）

```python
from services.features.sqli_worker import SQLiWorker
from services.aiva_common.schemas import FindingPayload
from services.aiva_common.enums import Severity, Confidence
from services.integration.aiva_integration.unified_data_manager import (
    get_unified_data_manager,
)
from uuid import uuid4
import logging

logger = logging.getLogger(__name__)

class SQLiWorker:
    """SQL 注入 Worker"""
    
    def __init__(self):
        self.integration_manager = get_unified_data_manager()
        self.payloads = self._load_payloads()
        self.current_scan_id = None
        logger.info("✅ SQLiWorker 已連接整合模組")
    
    async def test_sql_injection(self, target_url: str, param: str):
        """測試 SQL 注入漏洞"""
        
        findings = []
        
        # 步驟 1: 優化測試策略（基於歷史漏洞）
        await self.optimize_testing_strategy(target_url)
        
        # 步驟 2: 執行漏洞測試
        for payload in self.payloads:
            result = await self._test_payload(target_url, param, payload)
            
            if result.is_vulnerable:
                # 步驟 3: 創建漏洞發現
                finding = FindingPayload(
                    finding_id=f"sqli_{uuid4().hex[:8]}",
                    affected_url=target_url,
                    affected_parameter=param,
                    severity=Severity.CRITICAL,
                    confidence=Confidence.HIGH,
                    vulnerability_type="sql_injection",
                    description=f"SQL Injection in parameter '{param}'",
                    evidence={
                        "payload": payload,
                        "response": result.response[:500],  # 截斷長響應
                        "error_message": result.error,
                        "detection_method": result.detection_method,
                    },
                    cwe="CWE-89",
                    remediation="使用參數化查詢或 ORM",
                )
                
                findings.append(finding)
                
                # 步驟 4: 保存到整合模組
                self.integration_manager.save_finding(
                    finding=finding,
                    scan_id=self.current_scan_id,
                    task_id=f"sqli_test_{uuid4().hex[:8]}",
                )
                
                logger.info(f"✅ 發現 SQLi 漏洞: {finding.finding_id}")
        
        return findings
    
    async def optimize_testing_strategy(self, target_url: str):
        """基於歷史漏洞優化測試策略"""
        
        # 查詢歷史 SQLi 漏洞
        historical_sqli = self.integration_manager.query_findings(
            vulnerability_type="sql_injection",
            status="active",
            limit=100,
        )
        
        logger.info(f"載入 {len(historical_sqli)} 個歷史 SQLi 漏洞")
        
        if not historical_sqli:
            return
        
        # 分析相似目標
        similar_targets = [
            f for f in historical_sqli
            if self._is_similar_target(target_url, f.get("target_url"))
        ]
        
        if similar_targets:
            logger.info(f"發現 {len(similar_targets)} 個相似目標的歷史漏洞")
            
            # 提取成功的 Payload
            successful_payloads = []
            for finding in similar_targets:
                evidence = finding.get("raw_data", {}).get("evidence", {})
                payload = evidence.get("payload")
                if payload:
                    successful_payloads.append(payload)
            
            # 優先測試成功的 Payload
            if successful_payloads:
                self.payloads = successful_payloads + self.payloads
                logger.info(f"✅ 已添加 {len(successful_payloads)} 個歷史成功 Payload")
    
    def _is_similar_target(self, url1: str, url2: str) -> bool:
        """判斷兩個目標是否相似"""
        from urllib.parse import urlparse
        
        parsed1 = urlparse(url1)
        parsed2 = urlparse(url2)
        
        # 比較域名和路徑結構
        return (
            parsed1.netloc == parsed2.netloc or
            parsed1.path.split("/")[1:3] == parsed2.path.split("/")[1:3]
        )
    
    async def _test_payload(self, target_url: str, param: str, payload: str):
        """測試單個 Payload"""
        # 實際測試邏輯...
        pass
    
    def _load_payloads(self):
        """載入 Payload 列表"""
        return [
            "' OR '1'='1",
            "' OR 1=1--",
            "' UNION SELECT NULL--",
            # ... 更多 Payload
        ]
```

#### 關鍵要點

1. **歷史查詢**: 查詢歷史 SQLi 漏洞優化測試

```python
        return findings
```
